:::note
Esta página é gerada a partir de [`config/tweaks.json`](https://github.com/RamonMods/WinUtil/blob/main/config/tweaks.json). Não edite esta página diretamente.
:::

```json title="config/tweaks.json"
  "WPFTweaksDisableBitLocker": {
    "Content": "BitLocker - Desativar",
    "Description": "Desativa o BitLocker.",
    "category": "Essential Tweaks",
    "panel": "1",
    "InvokeScript": [
      "
      $bitLockerVolume = Get-BitLockerVolume -MountPoint $Env:SystemDrive -ErrorAction Stop

      if ($bitLockerVolume.VolumeStatus -eq 'FullyDecrypted') {
          Write-Host \"BitLocker is already disabled on $($Env:SystemDrive).\"
      } else {
          Disable-BitLocker -MountPoint $Env:SystemDrive -ErrorAction Stop
      }
      "
    ],
    "UndoScript": [
      "Enable-BitLocker -MountPoint $Env:SystemDrive"
    ],
  }
```
