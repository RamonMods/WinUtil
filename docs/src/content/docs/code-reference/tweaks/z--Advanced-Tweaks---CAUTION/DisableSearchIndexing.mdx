```json title="config/tweaks.json"
  "WPFTweaksDisableSearchIndexing": {
    "Content": "Indexação de Pesquisa do Windows - Desativar",
    "Description": "Para e desativa o serviço Windows Search (WSearch) para reduzir a atividade de disco em segundo plano, útil principalmente em HDs mecânicos. Aviso: as pesquisas de arquivos no Explorador e no Menu Iniciar ficam bem mais lentas, porque o Windows deixa de manter um índice de pesquisa. O desfazer restaura o serviço e o inicia novamente.",
    "category": "z__Advanced Tweaks - CAUTION",
    "panel": "1",
    "service": [
      {
        "Name": "WSearch",
        "StartupType": "Disabled",
        "OriginalType": "Automatic"
      }
    ],
    "InvokeScript": [
      "$search = Get-Service -Name WSearch -ErrorAction SilentlyContinue\n      if ($search -and $search.StartType -eq 'Disabled' -and $search.Status -ne 'Stopped') {\n        Stop-Service -Name WSearch -Force -ErrorAction Stop\n      }"
    ],
    "UndoScript": [
      "$search = Get-Service -Name WSearch -ErrorAction SilentlyContinue\n      if ($search -and $search.StartType -eq 'Automatic' -and $search.Status -ne 'Running') {\n        Start-Service -Name WSearch -ErrorAction Stop\n      }"
    ],
  }
```
