editUrl: false
---

:::note
Esta página é gerada a partir de [`config/tweaks.json`](https://github.com/RamonMods/WinUtil/blob/main/config/tweaks.json). Não edite esta página diretamente.
:::

```json title="config/tweaks.json"
  "WPFTweaksLogiBlock": {
    "Content": "Instalação automática do Logitech Download Assistant - Desativar",
    "Description": "Bloqueia o Logi Download Assistant, que o Windows Update fica reinstalando junto com os drivers de dispositivos Logitech. O hardware Logitech continua funcionando sem ele.",
    "category": "z__Advanced Tweaks - CAUTION",
    "panel": "1",
    "InvokeScript": [
      "
      Stop-Process -Name \"logi_download_assistant\" -Force -ErrorAction SilentlyContinue

      $ProgramFiles64 = if ($Env:ProgramW6432) { $Env:ProgramW6432 } else { $Env:ProgramFiles }
      $LogiPath = \"$ProgramFiles64\\LogiDownloadAssistant\"

      if (Test-Path $LogiPath) {
        Remove-Item $LogiPath\\* -Recurse -Force
      } else {
        New-Item -Path $LogiPath -ItemType Directory
      }

      icacls $LogiPath /deny \"*S-1-1-0:(W)\"
      if ($LASTEXITCODE -ne 0) { throw \"icacls failed to deny write access on $LogiPath (exit code $LASTEXITCODE)\" }
      "
    ],
    "UndoScript": [
      "
      $ProgramFiles64 = if ($Env:ProgramW6432) { $Env:ProgramW6432 } else { $Env:ProgramFiles }
      $LogiPath = \"$ProgramFiles64\\LogiDownloadAssistant\"

      if (Test-Path $LogiPath) {
        icacls $LogiPath /remove:d \"*S-1-1-0\"
        if ($LASTEXITCODE -ne 0) { throw \"icacls failed to remove the write-deny rule on $LogiPath (exit code $LASTEXITCODE)\" }
      }
      "
    ],
  }
```
