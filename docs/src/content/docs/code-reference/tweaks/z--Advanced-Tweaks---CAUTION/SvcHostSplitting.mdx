:::note
Esta página é gerada a partir de [`config/tweaks.json`](https://github.com/RamonMods/WinUtil/blob/main/config/tweaks.json). Não edite esta página diretamente.
:::

```json title="config/tweaks.json"
  "WPFTweaksSvcHostSplitting": {
    "Content": "Divisão de svchost.exe - Desativar",
    "Description": "Desativa a divisão do host de serviços para reduzir o número de processos svchost.exe e baixar um pouco o consumo de memória. Recomendado para sistemas de baixo desempenho com 12 GB de RAM ou menos. Pode interferir nos serviços do Xbox. Reinicie o Windows depois de alterar esta configuração.",
    "category": "z__Advanced Tweaks - CAUTION",
    "panel": "1",
    "InvokeScript": [
      "
      # A threshold above the physical memory disables svchost service splitting
      $Memory = (Get-CimInstance Win32_PhysicalMemory | Measure-Object Capacity -Sum).Sum / 1KB
      Set-ItemProperty -Path \"HKLM:\\SYSTEM\\CurrentControlSet\\Control\" -Name SvcHostSplitThresholdInKB -Value $Memory -Type DWord
      "
    ],
    "UndoScript": [
      "
      # Windows client default (0x380000): splitting back on
      Set-ItemProperty -Path \"HKLM:\\SYSTEM\\CurrentControlSet\\Control\" -Name SvcHostSplitThresholdInKB -Value 3670016 -Type DWord -Force
      "
    ],
  }
```
